stage('RabbitAITest 门禁') {
  steps {
    sh '''TASK=$(curl -s -u "$RABBIT_AK:$RABBIT_SK" -X POST .../open/exec/scenario \
           -d "{\"scenarioId\":\"$SCENARIO_ID\"}" | jq -r .data.taskId)
    for i in $(seq 1 60); do
      ST=$(curl -s -u "$RABBIT_AK:$RABBIT_SK" .../open/exec/$TASK | jq -r .data.status)
      [ "$ST" != "RUNNING" ] && break; sleep 5
    done
    curl -s -u "$RABBIT_AK:$RABBIT_SK" .../open/exec/$TASK/report | jq -e '.data.summary.failed == 0'
  }
}
